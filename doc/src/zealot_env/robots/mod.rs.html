1//! Robot asset specifications — the analogue of AGILE's `assets/robots/*.py`.
2//!
3//! A robot spec is **pure data**: the kinematic/actuation facts a task needs that
4//! don't belong to any single MDP — joint names and ordering, default ("home")
5//! pose, per-joint PD gains and effort limits, action scales, and where the
6//! MJCF/URDF live. It is deliberately free of nexus / GPU types so it builds and
7//! unit-tests without the `cargo-gpu` toolchain; the env loop (Isaac Lab
8//! "manager" tier) consumes it once physics is wired.
9//!
10//! All supported robots are 12-DOF lower-body walkers (6 per leg), so the
11//! observation/action layout is identical across them and [`RobotSpec`] is one
12//! concrete struct rather than a trait. Select at runtime with `BIPED_ROBOT`
13//! (see [`RobotSpec::from_env`]):
14//!
15//! - `lerobot` (default) — the LeRobot Humanoid bipedal platform.
16//! - `g1` — Unitree G1, the official legs-only 12-DOF model (upper body fused
17//!   into the pelvis), gains from unitree_rl_gym.
18//! - `h2plus` — Unitree H2 Plus, legs-only model generated from the official
19//!   URDF (no public RL config exists; gains are mass-scaled from the G1's).
20

22pub mod unitree_g1;
23pub mod unitree_h2_plus;

27/// Number of actuated leg DOFs (6 per leg) — shared by every supported robot.
28pub const NUM_JOINTS: usize = 12;
29
30/// Per-joint specification: gains, limits, action scale, and home pose.
31///
32/// One entry per actuated DOF. `kp`/`kd` are the position/velocity gains of the
33/// joint's PD controller (stiffness / damping); the env applies
34/// `τ = kp·(q_target − q) − kd·q̇`, saturated at `effort_limit`.
35#[derive(Clone, Copy, Debug, PartialEq)]
36pub struct JointSpec {
37    /// MJCF/URDF joint name (e.g. `"hipy_left"`, `"left_hip_pitch_joint"`).
38    pub name: &'static str,
39    /// Position gain (stiffness), N·m/rad.
40    pub kp: f32,
41    /// Velocity gain (damping), N·m·s/rad.
42    pub kd: f32,
43    /// Torque saturation, N·m.
44    pub effort_limit: f32,
45    /// Velocity limit, rad/s (soft-limit penalties reference this).
46    pub vel_limit: f32,
47    /// Action scale: `q_target = default_pos + scale · action`, action ∈ ~[-1, 1].
48    pub action_scale: f32,
49    /// Default ("home") joint position, rad.
50    pub default_pos: f32,
51    /// Hard joint position limits `(lower, upper)`, rad (from the model).
52    pub pos_limit: (f32, f32),
53    /// Rotor/reflected inertia (kg·m²), added to the joint's dof inertia — what
54    /// makes stiff PD-controlled joints numerically stable in sim.
55    pub armature: f32,
56    /// Passive joint damping (N·m·s/rad). Folded into the motor's velocity gain
57    /// since the nexus passive-damping buffer is otherwise a hardcoded 0.1
58    /// default. The MJCF `damping` attr, when present, takes precedence.
59    pub damping: f32,
60    /// Coulomb joint friction (N·m) — a constant torque opposing motion,
61    /// applied as `-frictionloss·sign(q̇)` via the nexus `dof_frictionloss`
62    /// buffer (energy dissipation / stiction).
63    pub frictionloss: f32,
64}

66/// A 12-DOF bipedal robot: joints in **canonical policy order** (the order in
67/// which joint quantities appear in observations and actions) plus the
68/// robot-level facts the env and task need. Pure `'static` data — every robot
69/// is a `const fn` constructor.
70#[derive(Clone, Copy, Debug)]
71pub struct RobotSpec {
72    /// Short identifier (`"lerobot"`, `"unitree_g1"`, `"unitree_h2_plus"`).
73    pub name: &'static str,
74    /// Per-joint specs, in canonical policy order.
75    pub joints: [JointSpec; NUM_JOINTS],
76    /// Root/base link name in the model.
77    pub base_link: &'static str,
78    /// Left / right foot link names (contact + foot-reward references). Must be
79    /// the links carrying the sole `class="collision"` capsules in the MJCF.
80    pub foot_links: [&'static str; 2],
81    /// The foot's "forward" direction in the foot link's LOCAL frame (used by
82    /// the foot-yaw-vs-base observation). +X for models whose foot frame is
83    /// upright; the Unitree G1's foot frame is axis-normalized so its forward
84    /// is +Z (see `scripts/convert_unitree_biped.py`).
85    pub foot_forward_local: [f32; 3],
86    /// Foot-link-origin height (m, world) below which the foot counts as in
87    /// ground contact. Robot-specific because the sole-to-link-origin offset
88    /// differs (lerobot/G1 ≈ 0.035 m, H2 Plus ≈ 0.054 m). `BIPED_CONTACT_Z`
89    /// overrides.
90    pub foot_contact_z: f32,
91    /// Base height in the DEFAULT (home) pose with the sole on the ground, m —
92    /// the base-height reward target.
93    pub base_height: f32,
94    /// Straight-leg (all joints at 0) base height with the sole exactly on the
95    /// ground, m — the spawn height (the multibody rest pose is q = 0).
96    pub spawn_z: f32,
97    /// Fall-termination floor on base height, m. Below this the episode ends —
98    /// this is what stops the policy reward-hacking by sinking the
99    /// (collider-less) torso through the ground while staying upright.
100    pub min_base_height: f32,
101    /// Total robot mass, kg (≈ sum of model link masses).
102    pub total_mass: f32,
103    /// MJCF path (zealot dialect — see `scripts/convert_unitree_biped.py`),
104    /// relative to `$HOME`.
105    pub mjcf_rel_path: &'static str,
106    /// Source URDF path, relative to `$HOME` (kept for the URDF-based smoke
107    /// examples and provenance).
108    pub urdf_rel_path: &'static str,
109    /// Left↔right mirror permutation over canonical joint indices:
110    /// `mirror[i]` is the contralateral counterpart of joint `i`.
111    pub mirror: [usize; NUM_JOINTS],
112    /// Mirror sign per joint: sagittal joints (pitch/knee) mirror equal (+1),
113    /// lateral joints (roll/yaw) mirror opposite (−1). Used by the bilateral
114    /// symmetry reward and the trainers' mirror-symmetry loss.
115    pub mirror_sign: [f32; NUM_JOINTS],
116    /// Canonical indices of the four hip yaw/roll DOFs (the jittery lateral
117    /// hips), for the targeted action-rate penalty.
118    pub hip_yawroll: [usize; 4],
119    /// Link-name fragments identifying the leg links that have NO ground
120    /// collider and must never legitimately touch the floor (WBC-AGILE-style
121    /// `illegal_contact` termination). Ankle + foot links must NOT match.
122    pub illegal_ground_fragments: &'static [&'static str],
123    /// Left/right link pairs (foot, shin, thigh) for the leg-crossing
124    /// self-collision guard: terminate if any pair gets closer than
125    /// `BIPED_SELF_COLL_DIST`.
126    pub self_collision_pairs: &'static [(&'static str, &'static str)],
127    /// PD gains for NON-action joints the model may carry (e.g. the G1
128    /// 29-DOF body's waist/arms, which the sim holds at the rest pose while
129    /// the policy drives the legs): `(name_fragment, kp, kd, effort_limit)`,
130    /// first matching fragment wins. Joints matching nothing fall back to the
131    /// env's generic holding gains. Empty for legs-only models.
132    pub held_joints: &'static [(&'static str, f32, f32, f32)],
133    /// Position target for held joints that must NOT sit at the model's rest
134    /// pose: `(name_fragment, target_rad)`, first matching fragment wins,
135    /// anything unmatched holds at 0 (the rest pose).
136    ///
137    /// The G1 MJCF bakes a **73.2 deg elbow bend into the body hierarchy**, so
138    /// q = 0 is NOT a straight arm -- it is the forearms held out in front,
139    /// like carrying a tray. The sim2sim harnesses and the LeRobot deploy
140    /// controller all command elbow = 1.28 rad, which CANCELS that built-in
141    /// bend and leaves the arm straight down along the body (1.1 deg of bend,
142    /// hand 0.179 m below the shoulder). Training at 0 therefore ran a
143    /// different upper-body geometry than every evaluator and the real robot.
144    pub held_home: &'static [(&'static str, f32)],
145}

147/// Programmatic override for [`RobotSpec::from_env`], for targets where
148/// environment variables don't exist (wasm web demos: `std::env::set_var`
149/// panics on wasm32-unknown-unknown). First set wins; call before the env is
150/// constructed.
151static ROBOT_OVERRIDE: std::sync::OnceLock<String> = std::sync::OnceLock::new();
152
153/// See [`ROBOT_OVERRIDE`]. Accepts the same names as `BIPED_ROBOT`.
154pub fn set_robot_override(name: &str) {
155    let _ = ROBOT_OVERRIDE.set(name.to_string());
156}

158impl RobotSpec {
159    /// Select the robot from `BIPED_ROBOT` (default: `lerobot`), then apply
160    /// `BIPED_ACTION_SCALE` if set.
161    ///
162    /// **The scale is easy to misread from the source.** The base
163    /// `unitree_g1()` spec says 0.25, but every G1 generation this campaign has
164    /// run uses `BIPED_ROBOT=g1_29dof_agile`, which chains through
165    /// `unitree_g1_agile()` — and that overwrites EVERY joint with 0.5 as one
166    /// of its AGILE deltas. So v19/v21/v24/v26 all trained at **0.5**, not the
167    /// 0.25 the base spec advertises. (This was mis-stated in the published
168    /// model card; 0.25 would halve every commanded joint excursion on
169    /// hardware.) Read `unitree_g1_agile()` before quoting a number.
170    ///
171    /// Lowering this is NOT a gentle version of the same problem: several terms
172    /// are denominated in ACTION units, not radians. `action_rate` penalises
173    /// (Δaction)², so at 0.25 the same physical motion costs 4x more; and
174    /// `init_noise_std`/`log_std` are action-space too, so exploration covers
175    /// half the physical range. Halving the scale without scaling `action_rate`
176    /// down ~4x changes the optimisation problem, not just the parameterisation
177    /// — and the policy could otherwise just emit 2x larger actions and land on
178    /// identical targets, leaving sensor-noise torque ripple unchanged.
179    pub fn from_env() -> Self {
180        // `ROBOT_OVERRIDE` first: the wasm demos cannot set env vars
181        // (`std::env::set_var` panics there), so they select the robot
182        // programmatically via [`set_robot_override`].
183        let name = ROBOT_OVERRIDE
184            .get()
185            .cloned()
186            .or_else(|| std::env::var("BIPED_ROBOT").ok())
187            .unwrap_or_default();
188        let mut spec = match name.as_str() {
189            // Default = the current production platform (was "lerobot").
190            "" | "g1_29dof_agile" => unitree_g1::unitree_g1_29dof_agile(),
191            "lerobot" => lerobot_bipedal::lerobot(),
192            "g1" | "unitree_g1" => unitree_g1::unitree_g1(),
193            "g1_agile" => unitree_g1::unitree_g1_agile(),
194            "g1_29dof" | "g1_29" | "g1full" => unitree_g1::unitree_g1_29dof(),
195            "h2plus" | "h2_plus" | "unitree_h2_plus" => unitree_h2_plus::unitree_h2_plus(),
196            other => {
197                panic!("unknown BIPED_ROBOT '{other}' (expected lerobot | g1 | g1_agile | g1_29dof | g1_29dof_agile | h2plus)")
198            }
199        };
200        if let Ok(v) = std::env::var("BIPED_ACTION_SCALE") {
201            if let Ok(s) = v.parse::<f32>() {
202                assert!(s > 0.0 && s <= 2.0, "BIPED_ACTION_SCALE {s} out of range (0, 2]");
203                for j in &mut spec.joints {
204                    j.action_scale = s;
205                }
206            }
207        }
208        spec
209    }

211    /// Default joint positions in canonical order (the home pose targets).
212    pub fn default_pose(&self) -> [f32; NUM_JOINTS] {
213        std::array::from_fn(|i| self.joints[i].default_pos)
214    }
215
216    /// Per-joint action scales in canonical order.
217    pub fn action_scales(&self) -> [f32; NUM_JOINTS] {
218        std::array::from_fn(|i| self.joints[i].action_scale)
219    }
220
221    /// Absolute path to the zealot-dialect MJCF.
222    ///
223    /// Resolution order:
224    /// 1. relative to the WORKSPACE ROOT (baked in at compile time) — this is
225    ///    what makes a fresh clone train without any setup: the converted
226    ///    MJCFs are committed under `assets/robots/`, nothing is downloaded;
227    /// 2. relative to the current directory (binaries moved off the build
228    ///    machine, run from the repo root);
229    /// 3. joined onto `$HOME` — the legacy hardcoded-layout form, and how the
230    ///    out-of-repo LeRobot asset is still found.
231    /// `BIPED_MJCF` (handled by the caller) overrides all of this.
232    pub fn mjcf_path(&self) -> std::path::PathBuf {
233        let workspace = std::path::Path::new(env!("CARGO_MANIFEST_DIR"))
234            .parent()
235            .unwrap_or_else(|| std::path::Path::new("."));
236        for base in [workspace, std::path::Path::new(".")] {
237            let p = base.join(self.mjcf_rel_path);
238            if p.exists() {
239                return p;
240            }
241        }
242        Self::home().join(self.mjcf_rel_path)
243    }
244
245    /// Absolute path to the source URDF, resolved against `$HOME`.
246    pub fn urdf_path(&self) -> std::path::PathBuf {
247        Self::home().join(self.urdf_rel_path)
248    }
249
250    fn home() -> std::path::PathBuf {
251        std::path::PathBuf::from(std::env::var("HOME").unwrap_or_else(|_| ".".to_string()))
252    }
253}

259    /// The in-repo robots must resolve to their committed MJCFs on a FRESH
260    /// CLONE — no `$HOME` layout assumptions, no downloads, no `BIPED_MJCF`.
261    /// (The LeRobot spec is excluded: its asset lives in a separate private
262    /// repo and is still resolved through the `$HOME` fallback.)
263    #[test]
264    fn committed_assets_resolve() {
265        for spec in [
266            unitree_g1::unitree_g1(),
267            unitree_g1::unitree_g1_29dof(),
268            unitree_g1::unitree_g1_29dof_agile(),
269            unitree_h2_plus::unitree_h2_plus(),
270        ] {
271            let p = spec.mjcf_path();
272            assert!(p.exists(), "{}: MJCF not found at {}", spec.name, p.display());
273        }
274    }

276    fn check_invariants(r: &RobotSpec) {
277        // Unique joint names.
278        let mut seen = std::collections::HashSet::new();
279        for j in &r.joints {
280            assert!(seen.insert(j.name), "{}: duplicate joint {}", r.name, j.name);
281        }
282        // Mirror is a sign-consistent involution pairing distinct joints.
283        for i in 0..NUM_JOINTS {
284            let m = r.mirror[i];
285            assert_ne!(m, i, "{}: joint {i} mirrors itself", r.name);
286            assert_eq!(r.mirror[m], i, "{}: mirror not an involution at {i}", r.name);
287            assert_eq!(
288                r.mirror_sign[i], r.mirror_sign[m],
289                "{}: mirror sign mismatch {i}<->{m}",
290                r.name
291            );
292        }
293        // hip_yawroll indices are valid and distinct.
294        let mut hy = std::collections::HashSet::new();
295        for &i in &r.hip_yawroll {
296            assert!(i < NUM_JOINTS);
297            assert!(hy.insert(i), "{}: duplicate hip_yawroll idx {i}", r.name);
298        }
299        // Default pose respects the joint limits; sane scalar facts.
300        for j in &r.joints {
301            assert!(j.pos_limit.0 <= j.default_pos && j.default_pos <= j.pos_limit.1);
302            assert!(j.kp > 0.0 && j.kd > 0.0 && j.effort_limit > 0.0);
303        }
304        assert!(r.min_base_height < r.base_height);
305        assert!(r.base_height <= r.spawn_z + 0.05); // home pose ≈ slightly crouched
306        assert!(r.total_mass > 1.0);
307    }

311        check_invariants(&lerobot_bipedal::lerobot());
312        check_invariants(&unitree_g1::unitree_g1());

314        check_invariants(&unitree_h2_plus::unitree_h2_plus());
