1//! THE config — every runtime knob that more than one crate/site consumes is
2//! declared here EXACTLY ONCE (env-var name + default + doc), parsed once,
3//! cached for the process lifetime.
4//!
5//! Motivation: knobs used to be `std::env::var` reads scattered per call site,
6//! each with its own copy of the default. When a default changed, every copy
7//! had to be found — and when one was missed, the two sites silently
8//! disagreed (v29 trained 16 h single-frame because the trainer's
9//! `BIPED_OBS_HISTORY` default said 5 while the env's said 1).
10//!
11//! Rules:
12//! - A knob read by MORE THAN ONE site MUST live here. Single-site knobs may
13//!   stay local, but never duplicate a default.
14//! - wasm/demo overrides: call [`Knob::set_override`] before first use (env
15//!   vars don't exist in the browser). Overrides beat env; env beats default.
16
17use std::str::FromStr;
18use std::sync::OnceLock;
19
20/// One runtime knob: env-var name, default, one-time parse + cache.
21pub struct Knob<T: 'static> {
22    name: &'static str,
23    default: T,
24    cell: OnceLock<T>,
25    ovr: OnceLock<T>,
26}
27
28impl<T: Copy + FromStr> Knob<T> {
29    pub const fn new(name: &'static str, default: T) -> Self {
30        Self {
31            name,
32            default,
33            cell: OnceLock::new(),
34            ovr: OnceLock::new(),
35        }
36    }
37
38    /// Resolve: override → env var → default. Cached on first call.
39    pub fn get(&self) -> T {
40        *self.cell.get_or_init(|| {
41            if let Some(v) = self.ovr.get() {
42                return *v;
43            }
44            std::env::var(self.name)
45                .ok()
46                .and_then(|s| s.parse().ok())
47                .unwrap_or(self.default)
48        })
49    }
50
51    /// Programmatic override (wasm demos). Must run before the first `get`.
52    pub fn set_override(&self, v: T) {
53        let _ = self.ovr.set(v);
54    }
55
56    pub fn name(&self) -> &'static str {
57        self.name
58    }
59}
60
61/// Boolean flags: default ON, `<name>=0` disables (the production convention).
62pub struct Flag {
63    name: &'static str,
64    default_on: bool,
65    cell: OnceLock<bool>,
66    ovr: OnceLock<bool>,
67}
68
69impl Flag {
70    pub const fn new(name: &'static str, default_on: bool) -> Self {
71        Self {
72            name,
73            default_on,
74            cell: OnceLock::new(),
75            ovr: OnceLock::new(),
76        }
77    }
78
79    pub fn get(&self) -> bool {
80        *self.cell.get_or_init(|| {
81            if let Some(v) = self.ovr.get() {
82                return *v;
83            }
84            match std::env::var(self.name).ok().as_deref() {
85                Some("0") => false,
86                Some(_) => true,
87                None => self.default_on,
88            }
89        })
90    }
91
92    pub fn set_override(&self, v: bool) {
93        let _ = self.ovr.set(v);
94    }
95}

98// The multi-site knobs (each was read in ≥2 places with per-site defaults).

102/// Actor observation frame stacking (1 = off). MUST be one value for the
103/// env's ring buffer AND the trainer's network sizing.
104pub static OBS_HISTORY: Knob<usize> = Knob::new("BIPED_OBS_HISTORY", 5);
105
106/// Rough-terrain difficulty curriculum.
107pub static TERRAIN: Flag = Flag::new("BIPED_TERRAIN", true);
108
109/// Force-based foot-contact sensing (feeds contact obs + force_rate reward).
110pub static CONTACT_SENSE: Flag = Flag::new("BIPED_CONTACT_SENSE", true);
111
112/// Per-pair contact-manifold reduction (≤4 points; the terrain perf lever).
113pub static CONTACT_REDUCE: Flag = Flag::new("BIPED_CONTACT_REDUCE", true);
114
115/// Contact-buffer pre-size per batch.
116pub static CONTACT_CAP: Knob<u32> = Knob::new("BIPED_CONTACT_CAP", 128);
117
118/// Contact natural frequency (Hz) — pinned production value.
119pub static CONTACT_NF: Knob<f32> = Knob::new("BIPED_CONTACT_NF", 240.0);
120
121/// Contact damping ratio — pinned production value.
122pub static CONTACT_DR: Knob<f32> = Knob::new("BIPED_CONTACT_DR", 1.0);
123
124/// Physics decimation (substeps per control step at fixed control_dt=0.02).
125pub static DECIMATION: Knob<u32> = Knob::new("BIPED_DECIMATION", 4);
126
127/// Substep refresh mode: 0 = refresh dynamics + constraints ONCE per control
128/// step (default), 1 = rebuild them every substep, 2 = light split cadence
129/// (constraints per substep, M/LU per step).
130///
131/// This is the single biggest physics-cost dial in the stack. Measured on a
132/// 5090 at N=4096 with the production config: mode 1 costs 114.6 ms of GPU per
133/// control step vs 22.5 ms at mode 0 — 4.4 s/iter vs 2.1 s/iter, i.e. per-substep
134/// refresh is ~5x the physics bill for the whole trainer.
135///
136/// The knob arrived with the nexus branch that added per-substep refresh and
137/// defaulted to 1, which silently halved throughput relative to v28 (2.5 s/iter,
138/// 62 ms/step) — that regression is what this default reverses.
139///
140/// NOTE: refresh cadence changes the physics slightly, so train and eval with
141/// the SAME setting; a policy trained at mode 1 is not guaranteed to transfer to
142/// mode 0 unchanged.
143pub static SUBSTEP_REFRESH: Knob<u32> = Knob::new("NEXUS_SUBSTEP_REFRESH", 0);

145#[cfg(test)]

149    #[test]
150    fn defaults_resolve_without_env() {
151        // The dangerous pair that diverged in v29: one source of truth now.
152        assert_eq!(OBS_HISTORY.get(), 5);
153        assert!(TERRAIN.get());
154        assert_eq!(DECIMATION.get(), 4);

157    #[test]
158    fn override_beats_default() {
159        static K: Knob<u32> = Knob::new("ZEALOT_TEST_KNOB_XYZ", 7);

161        assert_eq!(K.get(), 9);
