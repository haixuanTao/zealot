1//! Minimal vector / quaternion helpers.
2//!
3//! zealot-env builds with zero external dependencies (so it compiles without the
4//! `cargo-gpu` toolchain), so we can't lean on `glam`/`nalgebra` here. These are
5//! the handful of operations the MDP needs: rotating a world vector into the base
6//! frame and back, and reading a body axis out of a quaternion. Quaternions are
7//! `(x, y, z, w)` to match nexus's `Rotation` readback.
8
9/// 3-vector dot product.
10#[inline]
11pub fn dot(a: [f32; 3], b: [f32; 3]) -> f32 {
12    a[0] * b[0] + a[1] * b[1] + a[2] * b[2]
13}
14
15/// Rotate vector `v` by unit quaternion `q = (x, y, z, w)` (body→world if `q` is
16/// the body's orientation). Uses `v' = v + 2·q_xyz × (q_xyz × v + w·v)`.
17#[inline]
18pub fn quat_rotate(q: [f32; 4], v: [f32; 3]) -> [f32; 3] {
19    let u = [q[0], q[1], q[2]];
20    let w = q[3];
21    let t = cross(u, v);
22    let t = [t[0] + w * v[0], t[1] + w * v[1], t[2] + w * v[2]];
23    let tt = cross(u, t);
24    [v[0] + 2.0 * tt[0], v[1] + 2.0 * tt[1], v[2] + 2.0 * tt[2]]
25}
26
27/// Rotate `v` by the inverse of `q` (world→body for a body orientation `q`).
28#[inline]
29pub fn quat_rotate_inv(q: [f32; 4], v: [f32; 3]) -> [f32; 3] {
30    quat_rotate([-q[0], -q[1], -q[2], q[3]], v)
31}
32
33/// 3-vector cross product.
34#[inline]
35pub fn cross(a: [f32; 3], b: [f32; 3]) -> [f32; 3] {
36    [
37        a[1] * b[2] - a[2] * b[1],
38        a[2] * b[0] - a[0] * b[2],
39        a[0] * b[1] - a[1] * b[0],
40    ]
41}

43#[cfg(test)]
44mod tests {
45    use super::*;
46
47    #[test]
48    fn identity_quat_is_noop() {
49        let q = [0.0, 0.0, 0.0, 1.0];
50        let v = [1.0, 2.0, 3.0];
51        let r = quat_rotate(q, v);
52        for k in 0..3 {
53            assert!((r[k] - v[k]).abs() < 1e-6);
54        }
55    }
56
57    #[test]
58    fn rotate_then_inverse_recovers() {
59        // 90° about X: (x,y,z,w) = (sin45, 0, 0, cos45).
60        let s = std::f32::consts::FRAC_1_SQRT_2;
61        let q = [s, 0.0, 0.0, s];
62        let v = [0.0, 1.0, 0.0];
63        let r = quat_rotate(q, v);
64        // +Y rotated +90° about X → +Z.
65        assert!((r[0]).abs() < 1e-6 && (r[1]).abs() < 1e-6 && (r[2] - 1.0).abs() < 1e-6);
66        let back = quat_rotate_inv(q, r);
67        for k in 0..3 {
68            assert!((back[k] - v[k]).abs() < 1e-6);
69        }
70    }
