1//! PPO with a diagonal-Gaussian actor-critic — the `rsl_rl` tier.
2//!
3//! Matches the deployed policy's training config: separate actor/critic MLPs
4//! (ELU, e.g. `[512, 256, 128]`), a state-independent learnable `log_std`, GAE(λ)
5//! advantages, the clipped surrogate over K epochs × M minibatches, an
6//! adaptive-KL learning rate, an entropy bonus, clipped value loss, and global
7//! grad-norm clipping.
8//!
9//! The actor sees the (noised, partial) policy observation; the critic sees the
10//! privileged observation — the asymmetric setup AGILE uses. The update is a
11//! direct, allocation-light port of the math in `pendulum_ppo`, generalized to
12//! many action dims and a real value function.
13
14use crate::net::{Adam, Mlp, MlpGrad};
15use crate::rng::Lcg;
16use rayon::prelude::*;

18const LOG_SQRT_2PI: f32 = 0.918_938_5; // ½·ln(2π)

20/// Static PPO hyperparameters (the learning rate lives on [`ActorCritic`] because
21/// it adapts). Defaults are the deployed LeRobot policy's.
22#[derive(Clone, Copy, Debug)]
23pub struct PpoConfig {
24    /// Discount factor.
25    pub gamma: f32,
26    /// GAE λ.
27    pub lam: f32,
28    /// Surrogate clip ε.
29    pub clip: f32,
30    /// Entropy bonus coefficient.
31    pub entropy_coef: f32,
32    /// Value-loss coefficient.
33    pub value_coef: f32,
34    /// Learning epochs per update.
35    pub epochs: usize,
36    /// Minibatches per epoch.
37    pub minibatches: usize,
38    /// Global grad-norm clip.
39    pub max_grad_norm: f32,
40    /// Target KL for the adaptive LR schedule.
41    pub desired_kl: f32,
42    /// LR clamp `(min, max)` for the adaptive schedule.
43    pub lr_bounds: (f32, f32),
44    /// Whether to adapt the LR from the KL each minibatch (rsl_rl "adaptive"
45    /// schedule). When `false`, the LR is held fixed at its initial value.
46    pub adaptive_lr: bool,
47}
48
49impl Default for PpoConfig {
50    fn default() -> Self {
51        Self {

53            lam: 0.95,
54            clip: 0.2,
55            entropy_coef: 0.01,
56            value_coef: 1.0,
57            epochs: 5,
58            minibatches: 4,
59            max_grad_norm: 1.0,

104/// Running mean/variance normalizer (Welford), with a unit-variance prior so it
105/// is the identity until enough samples accumulate. rsl_rl's empirical
106/// observation normalization — applied to obs before the actor and to the
107/// privileged obs before the critic.
108#[derive(Clone, Debug)]
109pub struct Normalizer {
110    mean: Vec<f32>,
111    m2: Vec<f32>,
112    count: f32,
113}
114
115/// Welford state accumulated by `Normalizer::update_deferred`, merged into the
116/// live statistics by `Normalizer::commit` at an iteration boundary.
117#[derive(Clone, Debug, Default)]
118pub struct PendingNorm {
119    mean: Vec<f32>,
120    m2: Vec<f32>,
121    count: f32,
122}
123
124impl PendingNorm {
125    /// Fold one observation into the pending statistics (Welford).
126    pub fn push(&mut self, x: &[f32]) {
127        self.count += 1.0;
128        if self.mean.is_empty() {
129            self.mean = vec![0.0; x.len()];
130            self.m2 = vec![0.0; x.len()];
131        }
132        for i in 0..x.len() {
133            let d = x[i] - self.mean[i];
134            self.mean[i] += d / self.count;
135            self.m2[i] += d * (x[i] - self.mean[i]);
136        }
137    }
138}
139
140impl Normalizer {
141    /// New normalizer for `dim`-vectors (starts as the identity transform).
142    pub fn new(dim: usize) -> Self {
143        Self {
144            mean: vec![0.0; dim],
145            m2: vec![1.0; dim], // var = m2/count = 1 initially
146            count: 1.0,
147        }
148    }
149
150    /// Rebuild a normalizer directly from saved Welford state (used by `load`).
151    pub fn from_state(mean: Vec<f32>, m2: Vec<f32>, count: f32) -> Self {
152        Self { mean, m2, count }
153    }
154
155    /// Immutable view of the Welford state (used by `save`).
156    pub fn state(&self) -> (&[f32], &[f32], f32) {
157        (&self.mean, &self.m2, self.count)
158    }
159
160    /// Fold one observation into the running statistics.
161    pub fn update(&mut self, x: &[f32]) {
162        self.count += 1.0;
163        for i in 0..x.len() {
164            let d = x[i] - self.mean[i];
165            self.mean[i] += d / self.count;
166            self.m2[i] += d * (x[i] - self.mean[i]);
167        }
168    }
169
170    /// Merge a `PendingNorm` accumulated via `PendingNorm::push` (Chan et al.
171    /// parallel-Welford merge), then reset it. Lets a trainer freeze the
172    /// normalization transform for a whole rollout+update iteration (every
173    /// consumer — policy forward, stored `mean_old`/`logp_old`, GAE value
174    /// forwards, the update's staged obs, the mirror map — sees the SAME
175    /// transform), which makes the PPO importance ratios exact. Updating live
176    /// mid-iteration instead makes π_old's inputs unrecoverable and corrupts
177    /// the ratios (observed as a nonzero pre-update KL floor that grows when
178    /// obs statistics move fast).
179    pub fn commit(&mut self, pending: &mut PendingNorm) {
180        if pending.count == 0.0 {
181            return;
182        }
183        let n1 = self.count;
184        let n2 = pending.count;
185        let tot = n1 + n2;
186        for i in 0..self.mean.len() {
187            let d = pending.mean[i] - self.mean[i];
188            self.mean[i] += d * n2 / tot;
189            self.m2[i] += pending.m2[i] + d * d * n1 * n2 / tot;
190        }
191        self.count = tot;
192        pending.count = 0.0;
193        pending.mean.clear();
194        pending.m2.clear();
195    }
196
197    /// Whiten `x` to ~zero-mean/unit-variance, clamped to ±5.
198    pub fn normalize(&self, x: &[f32]) -> Vec<f32> {
199        (0..x.len())
200            .map(|i| {
201                let var = (self.m2[i] / self.count).max(1e-8);
202                ((x[i] - self.mean[i]) / var.sqrt()).clamp(-5.0, 5.0)
203            })
204            .collect()
205    }
206}

208/// A Gaussian actor-critic policy.
209#[derive(Clone, Debug)]
210pub struct ActorCritic {
211    /// Actor network: policy-obs → action mean.
212    pub actor: Mlp,
213    /// Critic network: privileged-obs → scalar value.
214    pub critic: Mlp,
215    /// State-independent log standard deviations (one per action dim).
216    pub log_std: Vec<f32>,
217    /// Running normalizer for the policy observation.
218    pub obs_norm: Normalizer,
219    /// Running normalizer for the privileged (critic) observation.
220    pub critic_norm: Normalizer,
221    opt_actor: Adam,
222    opt_critic: Adam,
223    // Adam state for log_std.
224    m_logstd: Vec<f32>,
225    v_logstd: Vec<f32>,
226    t_logstd: i32,
227    /// Current (adaptive) learning rate.
228    pub lr: f32,
229}

231impl ActorCritic {
232    /// Build from explicit layer stacks. `actor_dims[0]` is the policy-obs dim and
233    /// `actor_dims[last]` the action dim; `critic_dims[0]` is the privileged-obs
234    /// dim and `critic_dims[last]` must be 1.
235    pub fn new(
236        actor_dims: &[usize],
237        critic_dims: &[usize],
238        init_noise_std: f32,
239        init_lr: f32,
240        rng: &mut Lcg,
241    ) -> Self {
242        assert_eq!(
243            *critic_dims.last().unwrap(),
244            1,
245            "critic must output a scalar"
246        );
247        let action_dim = *actor_dims.last().unwrap();
248        // Small actor output init → gentle, near-default initial policy.
249        let actor = Mlp::new(actor_dims, 0.01, rng);
250        let critic = Mlp::new(critic_dims, 1.0, rng);
251        let opt_actor = Adam::new(&actor);
252        let opt_critic = Adam::new(&critic);
253        Self {
254            obs_norm: Normalizer::new(actor_dims[0]),
255            critic_norm: Normalizer::new(critic_dims[0]),
256            actor,
257            critic,
258            log_std: vec![init_noise_std.ln(); action_dim],
259            opt_actor,
260            opt_critic,
261            m_logstd: vec![0.0; action_dim],
262            v_logstd: vec![0.0; action_dim],
263            t_logstd: 0,
264            lr: init_lr,
265        }
266    }
267
268    /// Action dimensionality.
269    #[inline]
270    pub fn action_dim(&self) -> usize {
271        self.log_std.len()
272    }
273
274    /// Fold an observation pair into the running normalizers (call once per
275    /// collected step, during rollout).
276    pub fn record_obs(&mut self, obs: &[f32], critic_obs: &[f32]) {
277        self.obs_norm.update(obs);
278        self.critic_norm.update(critic_obs);
279    }

281    /// Actor mean for an observation (normalized internally).
282    pub fn mean(&self, obs: &[f32]) -> Vec<f32> {
283        self.actor
284            .forward(&self.obs_norm.normalize(obs))
285            .output()
286            .to_vec()
287    }
288
289    /// Critic value for a privileged observation (normalized internally).
290    pub fn value(&self, critic_obs: &[f32]) -> f32 {
291        self.critic
292            .forward(&self.critic_norm.normalize(critic_obs))
293            .output()[0]
294    }

296    /// Diagonal-Gaussian log-prob of `action` given `mean`.
297    pub fn logp(&self, action: &[f32], mean: &[f32]) -> f32 {
298        let mut lp = 0.0;
299        for k in 0..self.action_dim() {
300            let std = self.log_std[k].exp();
301            let d = (action[k] - mean[k]) / std;
302            lp += -0.5 * d * d - self.log_std[k] - LOG_SQRT_2PI;
303        }
304        lp
305    }
306
307    /// Policy entropy (nats).
308    pub fn entropy(&self) -> f32 {
309        let c = 0.5 + LOG_SQRT_2PI; // ½·ln(2πe)
310        self.log_std.iter().map(|ls| ls + c).sum()
311    }
312
313    /// Sample an action: `a = mean + std · ε`. Returns `(action, logp, mean)`.
314    pub fn sample(&self, obs: &[f32], rng: &mut Lcg) -> (Vec<f32>, f32, Vec<f32>) {
315        let mean = self.mean(obs);
316        let mut action = vec![0.0; self.action_dim()];
317        for k in 0..self.action_dim() {
318            let std = self.log_std[k].exp();
319            action[k] = mean[k] + std * rng.gauss();
320        }
321        let lp = self.logp(&action, &mean);
322        (action, lp, mean)
323    }
324
325    /// Exact KL `D(old ‖ new)` for diagonal Gaussians that share `log_std`
326    /// (so the std terms cancel): `½ Σ ((μ_new − μ_old)/σ)²`.
327    fn kl(&self, mean_new: &[f32], mean_old: &[f32]) -> f32 {
328        let mut kl = 0.0;
329        for k in 0..self.action_dim() {
330            let inv = (-self.log_std[k]).exp(); // 1/σ
331            kl += 0.5 * ((mean_new[k] - mean_old[k]) * inv).powi(2);
332        }
333        kl
334    }

336    /// Run the PPO update over `batch` (advantages/returns already filled). The
337    /// batch is normalized internally. Returns last-epoch diagnostics.
338    pub fn update(&mut self, batch: &mut [Sample], cfg: &PpoConfig) -> PpoStats {
339        let n = batch.len();
340        assert!(n >= cfg.minibatches, "fewer samples than minibatches");
341
342        // Normalize advantages across the batch.
343        let mean: f32 = batch.iter().map(|s| s.adv).sum::<f32>() / n as f32;
344        let var: f32 = batch.iter().map(|s| (s.adv - mean).powi(2)).sum::<f32>() / n as f32;
345        let sd = var.sqrt().max(1e-6);
346        for s in batch.iter_mut() {
347            s.adv = (s.adv - mean) / sd;
348        }
349
350        let mb = n / cfg.minibatches;
351        let mut stats = PpoStats {
352            lr: self.lr,
353            ..Default::default()
354        };
355
356        for _ in 0..cfg.epochs {
357            // A fresh shuffle each epoch (Fisher–Yates with the same LCG).
358            let mut order: Vec<usize> = (0..n).collect();
359            let mut sh = Lcg::new(self.t_logstd as u64 + 1);
360            for i in (1..n).rev() {
361                let j = (sh.unit() * (i + 1) as f32) as usize;
362                order.swap(i, j.min(i));
363            }
364
365            let mut epoch = PpoStats::default();
366            for m in 0..cfg.minibatches {
367                let slice = &order[m * mb..(m + 1) * mb];
368                let s = self.minibatch_step(batch, slice, cfg);
369                epoch.kl += s.kl;
370                epoch.surrogate += s.surrogate;
371                epoch.value_loss += s.value_loss;
372                epoch.entropy += s.entropy;
373            }
374            let inv = 1.0 / cfg.minibatches as f32;
375            stats = PpoStats {
376                kl: epoch.kl * inv,
377                surrogate: epoch.surrogate * inv,
378                value_loss: epoch.value_loss * inv,
379                entropy: epoch.entropy * inv,
380                lr: self.lr,
381            };
382        }
383        stats
384    }

386    /// One minibatch: accumulate gradients (data-parallel across cores), adapt LR
387    /// from the KL, clip, and step.
388    fn minibatch_step(&mut self, batch: &[Sample], idx: &[usize], cfg: &PpoConfig) -> PpoStats {
389        let action_dim = self.action_dim();
390        let scale = 1.0 / idx.len() as f32;
391
392        // Per-job accumulator: (actor grad, critic grad, log_std grad, surrogate,
393        // value_loss, kl). Folded over disjoint sample chunks in parallel — every
394        // read of `self` (forward/backward/logp/kl) is immutable, so this is sound.
395        type Acc = (MlpGrad, MlpGrad, Vec<f32>, f32, f32, f32);
396        let init = || {
397            (
398                MlpGrad::zero(&self.actor),
399                MlpGrad::zero(&self.critic),
400                vec![0.0f32; action_dim],
401                0.0f32,
402                0.0f32,
403                0.0f32,
404            )
405        };
406        let (g_actor, g_critic, g_logstd, surr_sum, vloss_sum, kl_sum) = idx
407            .par_iter()
408            .fold(init, |mut acc, &i| {
409                let s = &batch[i];
410                // --- actor (normalized obs) ---
411                let act = self.actor.forward(&self.obs_norm.normalize(&s.obs));
412                let mean = act.output().to_vec();
413                let logp = self.logp(&s.action, &mean);
414                let ratio = (logp - s.logp_old).exp();
415                let a = s.adv;
416                let surr = (ratio * a).min(ratio.clamp(1.0 - cfg.clip, 1.0 + cfg.clip) * a);
417                acc.3 += surr * scale;
418
419                let clipped =
420                    (a >= 0.0 && ratio > 1.0 + cfg.clip) || (a < 0.0 && ratio < 1.0 - cfg.clip);
421                let mut g_mean = vec![0.0f32; action_dim];
422                for k in 0..action_dim {
423                    let inv_var = (-2.0 * self.log_std[k]).exp(); // 1/σ²
424                    if !clipped {
425                        let d = s.action[k] - mean[k];
426                        g_mean[k] = -(a * ratio * d * inv_var) * scale;
427                        let dls = a * ratio * ((d * d) * inv_var - 1.0);
428                        acc.2[k] += -dls * scale;
429                    }
430                    // Entropy bonus: −entropy_coef·entropy, d/d log_std = −entropy_coef.
431                    acc.2[k] += -cfg.entropy_coef * scale;
432                }
433                self.actor.backward(&act, &g_mean, &mut acc.0);
434
435                // --- critic (clipped value loss, normalized obs) ---
436                let vact = self
437                    .critic
438                    .forward(&self.critic_norm.normalize(&s.critic_obs));
439                let v = vact.output()[0];
440                let v_clipped = s.value_old + (v - s.value_old).clamp(-cfg.clip, cfg.clip);
441                let l_unclipped = (v - s.ret).powi(2);
442                let l_clipped = (v_clipped - s.ret).powi(2);
443                let (vloss, dv) = if l_clipped > l_unclipped {
444                    (l_clipped, 2.0 * (v_clipped - s.ret))
445                } else {
446                    (l_unclipped, 2.0 * (v - s.ret))
447                };
448                acc.4 += vloss * scale;
449                let gv = [cfg.value_coef * dv * scale];
450                self.critic.backward(&vact, &gv, &mut acc.1);
451
452                acc.5 += self.kl(&mean, &s.mean_old) * scale;
453                acc
454            })
455            .reduce(init, |mut a: Acc, b: Acc| {
456                a.0.add(&b.0);
457                a.1.add(&b.1);
458                for k in 0..action_dim {
459                    a.2[k] += b.2[k];
460                }
461                (a.0, a.1, a.2, a.3 + b.3, a.4 + b.4, a.5 + b.5)
462            });
463
464        let mut g_actor = g_actor;
465        let mut g_critic = g_critic;
466        let g_logstd = g_logstd;
467        let mut acc = PpoStats {
468            surrogate: surr_sum,
469            value_loss: vloss_sum,
470            kl: kl_sum,
471            ..Default::default()
472        };
473        acc.entropy = self.entropy();
474
475        // Adaptive LR from the KL (rsl_rl rule), applied before stepping.
476        if cfg.adaptive_lr {
477            if acc.kl > cfg.desired_kl * 2.0 {
478                self.lr = (self.lr / 1.5).max(cfg.lr_bounds.0);
479            } else if acc.kl > 0.0 && acc.kl < cfg.desired_kl / 2.0 {
480                self.lr = (self.lr * 1.5).min(cfg.lr_bounds.1);
481            }
482        }
483
484        // Clip and step.
485        g_actor.clip(cfg.max_grad_norm);
486        g_critic.clip(cfg.max_grad_norm);
487        self.opt_actor.step(&mut self.actor, &g_actor, self.lr);
488        self.opt_critic.step(&mut self.critic, &g_critic, self.lr);
489        self.step_log_std(&g_logstd);
490
491        acc.lr = self.lr;
492        acc
493    }

495    /// Adam step for the `log_std` parameter vector.
496    fn step_log_std(&mut self, g: &[f32]) {
497        self.t_logstd += 1;
498        let (b1, b2, eps) = (0.9f32, 0.999f32, 1e-8f32);
499        let bc1 = 1.0 - b1.powi(self.t_logstd);
500        let bc2 = 1.0 - b2.powi(self.t_logstd);
501        for k in 0..self.action_dim() {
502            self.m_logstd[k] = b1 * self.m_logstd[k] + (1.0 - b1) * g[k];
503            self.v_logstd[k] = b2 * self.v_logstd[k] + (1.0 - b2) * g[k] * g[k];
504            let step = self.lr * (self.m_logstd[k] / bc1) / ((self.v_logstd[k] / bc2).sqrt() + eps);
505            // Keep std in a sane band so exploration neither vanishes nor explodes.
506            // Floor ≈ ln(0.2): a std of 0.2 keeps enough exploration to escape
507            // local optima (e.g. the sink), rather than collapsing to ~0.
508            self.log_std[k] = (self.log_std[k] - step).clamp(-1.6f32, 1.0);
509        }
510    }

512    /// Serialise the policy to a `.safetensors` file — HuggingFace's standard
513    /// model-weights format. Loadable cross-language from Python via the
514    /// `safetensors` package, with type+shape metadata in the JSON header.
515    /// Saves actor + critic networks, `log_std`, both Welford normalizers, and
516    /// `lr`. Adam optimizer state is NOT saved (re-initialised on load), so the
517    /// file is evaluation-ready but resuming training loses adaptive momentum.
518    ///
519    /// Tensor layout (all `f32`):
520    /// - `actor.w_{l}`  shape `[out, in]`, row-major (per-layer, l = 0..L-1)
521    /// - `actor.b_{l}`  shape `[out]`
522    /// - `critic.w_{l}` / `critic.b_{l}` — same pattern
523    /// - `log_std`            shape `[act_dim]`
524    /// - `obs_norm.mean`      shape `[obs_dim]`
525    /// - `obs_norm.m2`        shape `[obs_dim]`
526    /// - `obs_norm.count`     shape `[1]`
527    /// - `critic_norm.mean` / `.m2` / `.count` — same pattern
528    /// - `lr`                  shape `[1]`
529    pub fn save(&self, path: &str) -> std::io::Result<()> {
530        use safetensors::tensor::{Dtype, TensorView};
531        // Scalars need to live until after `serialize` so the views' borrows
532        // stay valid — bind them up-front.
533        let (obs_mean, obs_m2, obs_count) = self.obs_norm.state();
534        let (crit_mean, crit_m2, crit_count) = self.critic_norm.state();
535        let obs_count_arr = [obs_count];
536        let crit_count_arr = [crit_count];
537        let lr_arr = [self.lr];
538
539        let mut tensors: Vec<(String, TensorView)> = Vec::new();
540        // Actor + critic layers.
541        for (prefix, mlp) in [("actor", &self.actor), ("critic", &self.critic)] {
542            for l in 0..mlp.dims.len() - 1 {
543                let (in_d, out_d) = (mlp.dims[l], mlp.dims[l + 1]);
544                tensors.push((
545                    format!("{prefix}.w_{l}"),
546                    TensorView::new(Dtype::F32, vec![out_d, in_d], f32_bytes(&mlp.w[l]))
547                        .map_err(io_err)?,
548                ));
549                tensors.push((
550                    format!("{prefix}.b_{l}"),
551                    TensorView::new(Dtype::F32, vec![out_d], f32_bytes(&mlp.b[l]))
552                        .map_err(io_err)?,
553                ));
554            }
555        }
556        // 1-D tensors.
557        for (name, slice) in [
558            ("log_std", self.log_std.as_slice()),
559            ("obs_norm.mean", obs_mean),
560            ("obs_norm.m2", obs_m2),
561            ("obs_norm.count", obs_count_arr.as_slice()),
562            ("critic_norm.mean", crit_mean),
563            ("critic_norm.m2", crit_m2),
564            ("critic_norm.count", crit_count_arr.as_slice()),
565            ("lr", lr_arr.as_slice()),
566        ] {
567            tensors.push((
568                name.into(),
569                TensorView::new(Dtype::F32, vec![slice.len()], f32_bytes(slice)).map_err(io_err)?,
570            ));
571        }
572
573        let bytes = safetensors::serialize(tensors, &None).map_err(io_err)?;
574        std::fs::write(path, &bytes)
575    }

577    /// Load a policy previously written by [`save`]. Adam state is reinitialised.
578    pub fn load(path: &str) -> std::io::Result<Self> {
579        let bytes = std::fs::read(path)?;
580        Self::load_from_bytes(&bytes)
581    }

583    /// [`load`], but from an in-memory `.safetensors` image — for targets with
584    /// no filesystem (wasm demos embed the checkpoint via `include_bytes!`).
585    pub fn load_from_bytes(bytes: &[u8]) -> std::io::Result<Self> {
586        use safetensors::SafeTensors;
587        let st = SafeTensors::deserialize(bytes).map_err(io_err)?;
588        // Read an Mlp by scanning `<prefix>.w_{l}` and `<prefix>.b_{l}` for
589        // contiguous l starting at 0. `dims` is reconstructed from the shapes.
590        let read_mlp = |prefix: &str| -> std::io::Result<Mlp> {
591            let mut w = Vec::new();
592            let mut b = Vec::new();
593            let mut dims: Vec<usize> = Vec::new();
594            for l in 0usize.. {
595                let wn = format!("{prefix}.w_{l}");
596                let bn = format!("{prefix}.b_{l}");
597                let Ok(tw) = st.tensor(&wn) else { break };
598                let tb = st.tensor(&bn).map_err(io_err)?;
599                let sh = tw.shape();
600                if sh.len() != 2 {
601                    return Err(io_err_msg(format!("{wn}: expected 2D, got {sh:?}")));
602                }
603                if l == 0 {
604                    dims.push(sh[1]);
605                }
606                dims.push(sh[0]);
607                w.push(bytes_to_f32(tw.data()));
608                b.push(bytes_to_f32(tb.data()));
609            }
610            if w.is_empty() {
611                return Err(io_err_msg(format!("no layers under prefix `{prefix}`")));
612            }
613            Ok(Mlp { dims, w, b })
614        };
615        let actor = read_mlp("actor")?;
616        let critic = read_mlp("critic")?;
617        let read_vec = |name: &str| -> std::io::Result<Vec<f32>> {
618            Ok(bytes_to_f32(st.tensor(name).map_err(io_err)?.data()))
619        };
620        let read_scalar = |name: &str| -> std::io::Result<f32> { Ok(read_vec(name)?[0]) };
621        let log_std = read_vec("log_std")?;
622        let obs_norm = Normalizer::from_state(
623            read_vec("obs_norm.mean")?,
624            read_vec("obs_norm.m2")?,
625            read_scalar("obs_norm.count")?,
626        );
627        let critic_norm = Normalizer::from_state(
628            read_vec("critic_norm.mean")?,
629            read_vec("critic_norm.m2")?,
630            read_scalar("critic_norm.count")?,
631        );
632        let lr = read_scalar("lr")?;
633        let opt_actor = Adam::new(&actor);
634        let opt_critic = Adam::new(&critic);
635        let m_logstd = vec![0.0; log_std.len()];
636        let v_logstd = vec![0.0; log_std.len()];
637        Ok(Self {
638            actor,
639            critic,
640            log_std,
641            obs_norm,
642            critic_norm,
643            opt_actor,
644            opt_critic,
645            m_logstd,
646            v_logstd,
647            t_logstd: 0,
648            lr,
649        })
650    }
651}

655/// Reinterpret an `f32` slice as the equivalent little-endian byte slice. Safe
656/// on every platform Rust targets in practice (`f32` has fixed 4-byte layout).
657fn f32_bytes(v: &[f32]) -> &[u8] {
658    unsafe { std::slice::from_raw_parts(v.as_ptr() as *const u8, std::mem::size_of_val(v)) }
659}
660
661/// Inverse of [`f32_bytes`]: decode a little-endian byte slice into `f32`s.
662fn bytes_to_f32(b: &[u8]) -> Vec<f32> {
663    let mut out = Vec::with_capacity(b.len() / 4);
664    for chunk in b.chunks_exact(4) {
665        out.push(f32::from_le_bytes(chunk.try_into().unwrap()));
666    }
667    out
668}

670fn io_err<E: std::fmt::Debug>(e: E) -> std::io::Error {
671    std::io::Error::new(std::io::ErrorKind::InvalidData, format!("{e:?}"))
672}

678/// GAE(λ) advantages and returns for one trajectory (`rewards`, `values`,
679/// `dones` are per-step; `last_value` bootstraps the final step). `dones[t]`
680/// being true zeroes the bootstrap across that boundary.
681pub fn gae(
682    rewards: &[f32],
683    values: &[f32],
684    dones: &[bool],
685    last_value: f32,
686    gamma: f32,
687    lam: f32,
688) -> (Vec<f32>, Vec<f32>) {
689    let t = rewards.len();
690    let mut adv = vec![0.0f32; t];
691    let mut ret = vec![0.0f32; t];
692    let mut gae_acc = 0.0;
693    let mut next_v = last_value;
694    for i in (0..t).rev() {
695        let nonterminal = if dones[i] { 0.0 } else { 1.0 };
696        let delta = rewards[i] + gamma * next_v * nonterminal - values[i];
697        gae_acc = delta + gamma * lam * nonterminal * gae_acc;
698        adv[i] = gae_acc;
699        ret[i] = gae_acc + values[i];
700        next_v = values[i];
701    }
702    (adv, ret)
703}

723    #[test]
724    fn actor_critic_save_load_roundtrips() {
725        // Build a small actor-critic, feed in some normalizer updates, then save +
726        // load and verify the loaded policy produces bit-identical actions and the
727        // loaded normalizer produces bit-identical normalize() output.
728        let mut rng = Lcg::new(7);
729        let mut ac = ActorCritic::new(&[4, 8, 8, 2], &[6, 8, 1], 0.3, 1e-3, &mut rng);
730        // Push some data through the normalizers so their state isn't trivial.
731        for k in 0..20 {
732            let obs: Vec<f32> = (0..4).map(|j| ((k + j) as f32).sin()).collect();
733            let cobs: Vec<f32> = (0..6).map(|j| ((k * 3 + j) as f32).cos()).collect();
734            ac.record_obs(&obs, &cobs);
735        }
736        // Capture reference outputs at a few inputs.
737        let probes: Vec<Vec<f32>> = (0..3)
738            .map(|k| (0..4).map(|j| ((k * 7 + j) as f32).tanh()).collect())
739            .collect();
740        let before: Vec<Vec<f32>> = probes.iter().map(|p| ac.mean(p)).collect();
741        // Save → load.
742        let path = format!("/tmp/zealot_rl_roundtrip_{}.bin", std::process::id());
743        ac.save(&path).expect("save");
744        let ac2 = ActorCritic::load(&path).expect("load");
745        std::fs::remove_file(&path).ok();
746        // Loaded policy should produce IDENTICAL outputs (no numeric drift —
747        // weights are byte-for-byte the same).
748        for (probe, ref_out) in probes.iter().zip(before.iter()) {
749            let after = ac2.mean(probe);
750            assert_eq!(after.len(), ref_out.len());
751            for (a, b) in after.iter().zip(ref_out.iter()) {

753            }
754        }
755        // log_std and lr survive.
756        assert_eq!(ac.log_std, ac2.log_std);
757        assert_eq!(ac.lr, ac2.lr);
758    }

770    /// A 1-step Gaussian bandit: reward = −‖action − target‖². PPO should pull the
771    /// actor mean toward the target and raise mean reward.
772    #[test]
773    fn ppo_improves_on_bandit() {
774        let mut rng = Lcg::new(2024);
775        let obs_dim = 3;
776        let act_dim = 2;
777        let target = [0.7f32, -0.4];
778        let mut ac = ActorCritic::new(
779            &[obs_dim, 32, act_dim],
780            &[obs_dim, 32, 1],
781            0.6,
782            3e-3,
783            &mut rng,
784        );
785        let cfg = PpoConfig {
786            entropy_coef: 0.0,
787            minibatches: 4,
788            epochs: 4,
789            ..Default::default()
790        };
791        let obs = [0.5f32, -0.2, 0.1];
792
793        let mean_reward = |ac: &ActorCritic| -> f32 {
794            let m = ac.mean(&obs);
795            -((m[0] - target[0]).powi(2) + (m[1] - target[1]).powi(2))
796        };
797        let r0 = mean_reward(&ac);
798
799        for _ in 0..120 {
800            let mut batch: Vec<Sample> = Vec::new();
801            let n = 256;
802            let mut rewards = Vec::new();
803            let mut values = Vec::new();
804            for _ in 0..n {
805                let (action, logp, mean_old) = ac.sample(&obs, &mut rng);
806                let r = -((action[0] - target[0]).powi(2) + (action[1] - target[1]).powi(2));
807                let v = ac.value(&obs);
808                rewards.push(r);
809                values.push(v);
810                batch.push(Sample {
811                    obs: obs.to_vec(),
812                    critic_obs: obs.to_vec(),
813                    action,
814                    mean_old,
815                    logp_old: logp,
816                    value_old: v,
817                    adv: 0.0,
818                    ret: 0.0,
819                });
820            }
821            // Each sample is its own 1-step episode.
822            for i in 0..n {
823                let (a, ret) = gae(
824                    &[rewards[i]],
825                    &[values[i]],
826                    &[true],
827                    0.0,
828                    cfg.gamma,
829                    cfg.lam,
830                );
831                batch[i].adv = a[0];
832                batch[i].ret = ret[0];
833            }
834            ac.update(&mut batch, &cfg);
835        }
836
837        let r1 = mean_reward(&ac);
838        assert!(
839            r1 > r0 + 0.2,
840            "PPO did not improve bandit reward: {r0} -> {r1}"
841        );
842        let m = ac.mean(&obs);
843        assert!((m[0] - target[0]).abs() < 0.25, "mean[0] off: {}", m[0]);
844        assert!((m[1] - target[1]).abs() < 0.25, "mean[1] off: {}", m[1]);
845    }
